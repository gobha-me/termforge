# 22headers: compile every public header standalone with ONLY the public
# include path (issue #54). The generic per-dir rule in test/CMakeLists.txt
# only fires when a dir has no CMakeLists.txt, so this file takes over: it
# generates one TU per header and builds them with ${CMAKE_SOURCE_DIR}/include
# as the sole include dir. A public header that includes a private one fails
# the BUILD here.

file(GLOB_RECURSE PUBLIC_HEADERS ${CMAKE_SOURCE_DIR}/include/termforge/*.hpp)

set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})

set(HEADER_TUS "")
foreach(HDR ${PUBLIC_HEADERS})
  file(RELATIVE_PATH REL ${CMAKE_SOURCE_DIR}/include ${HDR})
  string(MAKE_C_IDENTIFIER ${REL} IDENT)
  set(TU ${GEN_DIR}/header_${IDENT}.cpp)
  file(WRITE ${TU} "// Generated by 22headers: standalone-compile one public header.\n#include <${REL}>\n")
  list(APPEND HEADER_TUS ${TU})
endforeach()

add_executable(22headers-test
  ${CMAKE_CURRENT_LIST_DIR}/test.cpp
  ${HEADER_TUS}
)

# The ONLY include path a library consumer is promised.
target_include_directories(22headers-test PRIVATE ${CMAKE_SOURCE_DIR}/include)

target_link_libraries(22headers-test PRIVATE Catch2::Catch2WithMain)
